cmake_minimum_required(VERSION 3.20)
project(KBMAudioCapture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build with MSVC and a Windows SDK supporting process loopback capture")
endif()
set(WIL_INCLUDE_DIR "" CACHE PATH "include directory from Microsoft.Windows.ImplementationLibrary 1.0.210204.1")
if(NOT EXISTS "${WIL_INCLUDE_DIR}/wil/com.h")
    message(FATAL_ERROR "Set WIL_INCLUDE_DIR to the NuGet package's include directory")
endif()
add_executable(KBMAudioCapture ApplicationLoopback.cpp LoopbackCapture.cpp MicCapture.cpp)
target_compile_features(KBMAudioCapture PRIVATE cxx_std_17)
target_compile_definitions(KBMAudioCapture PRIVATE UNICODE _UNICODE)
target_include_directories(KBMAudioCapture PRIVATE "${WIL_INCLUDE_DIR}")
target_link_libraries(KBMAudioCapture PRIVATE ole32 oleaut32 uuid mfplat mmdevapi mfuuid
    mfreadwrite windowsapp userenv winmm)
